
SELECT 
b.id
,b.title
,u.id as "user_id"
,u.username
FROM t_user as u
JOIN t_board as b ON u.id = b.user_id
;
-- INNER JOIN: 짝이 맞는 데이터만 나옴
-- 짝이 안맞은 데이터는 안나옴 


SELECT 
u.id as "user_id"
,u.username
,b.id as "board_id"
,b.title
FROM t_user as u 
LEFT JOIN t_board as b ON u.id = b.user_id
;
-- t_user 테이블에 있는 데이터는 다 나와야되. 짝이 안맞으면 NULL로 보여줘

SELECT 
u.id as "user_id"
,u.username
,b.id as "board_id"
,b.title
FROM t_board as b 
RIGHT JOIN t_user as u ON u.id = b.user_id
;
-- t_user 테이블에 있는 데이터는 다 나와야되. 짝이 안맞으면 NULL로 보여줘
