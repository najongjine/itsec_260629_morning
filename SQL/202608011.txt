CREATE TABLE t_user(
 id SERIAL PRIMARY KEY,
 username VARCHAR NOT NULL UNIQUE,
 password VARCHAR NOT NULL,
 email VARCHAR,
 gender VARCHAR CHECK (gender IN ('m', 'f')),
 created_dt TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE t_board(
 id SERIAL PRIMARY KEY,
 title VARCHAR,
 content TEXT,
 created_dt TIMESTAMPTZ DEFAULT NOW(),
 user_id INT,
 CONSTRAINT fk_user
 FOREIGN KEY (user_id)
 REFERENCES t_user(id)
 ON DELETE CASCADE
 ON UPDATE CASCADE
);

CREATE TABLE t_comment(
 id SERIAL PRIMARY KEY,
 cmt TEXT,
 created_dt TIMESTAMPTZ DEFAULT NOW(),
 user_id INT,
 board_id INT,
 CONSTRAINT fk_cmt_user
	 FOREIGN KEY (user_id)
	 REFERENCES t_user(id)
	 ON DELETE CASCADE
	 ON UPDATE CASCADE,
 CONSTRAINT fk_cmt_board
	 FOREIGN KEY (board_id)
	 REFERENCES t_board(id)
	 ON DELETE CASCADE
	 ON UPDATE CASCADE
);
