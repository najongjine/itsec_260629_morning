SELECT -- 2번째 실핼
t.id as "유저아이디"
, t.id * t.id AS "테이블엔없는칼럼"
,t.username as "유저네임"
FROM t_user AS t -- 1번째 실행
;

-- AS 는 별명 붙이기


SELECT
DISTINCT (c.board_id) AS "게시글을쓴_유저목록"
FROM t_comment as c
;
-- DISTINCT 는 진짜 중복 제거


SELECT -- 3번째 실행
*
FROM t_user AS u -- 1번째 실행
WHERE u.gender = 'f' -- 2번째 실행
AND u.created_dt < '2026-08-13'
;

SELECT
*
FROM t_board AS b
WHERE b.title LIKE '%첫%'
;
-- LIKE 는 문자열 부분검색 할때 사용됨
-- 속도는 느림

SELECT
*
FROM t_board AS b
WHERE b.title = '첫'
;
-- =은 완전일치에 사용됨
-- 속도는 빠름


SELECT
*
FROM t_board AS b
WHERE b.user_id = 2 OR b.user_id = 4 OR b.user_id = 1
;
-- OR 뜻은 조건 만족되는거 다 보여줘

SELECT
*
FROM t_board AS b
WHERE b.user_id IN(1,2,3,4)
;
-- IN 이랑 OR이랑 같다. 그런데 IN의 장점은 쿼리조건이 훨씬 짧아진다


SELECT
* 
FROM t_board
WHERE created_dt BETWEEN '2026-08-08' AND '2026-08-12'
;
-- 2026-08-08 이상 ~ 2026-08-12 이하



SELECT
*
FROM t_user 
WHERE email IS NULL
;
-- NULL은 =, != 로 검색하면 결과가 안나옴. 꼭 IS NULL, IS NOT NULL 로 검색 해야함


SELECT
*
FROM t_board
ORDER BY created_dt DESC
LIMIT 2
;
-- ORDER BY 는 정렬. ASC(디폴트) 오름차순, DESC 내림차순
-- LIMIT 결과 튜플의 수를 제한 하는것
